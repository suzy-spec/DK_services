/** * DK Analytics Dashboard ↔ Google Sheets 백엔드 (v8) * * v7 대비 변경점: * 1) 링크드인 채널 완전 삭제 * 2) 인기 카테고리/콘텐츠용 "스냅샷" 시트 3개 추가 * - 홈페이지_인기카테고리 : 날짜 | 카테고리 | 조회수 * - 홈페이지_인기콘텐츠 : 날짜 | 콘텐츠명 | 조회수 * - 네이버블로그_인기콘텐츠 : 날짜 | 콘텐츠명 | 조회수 * → GA4에서 확인할 때마다(비정기, 최소 주1회) 그 날짜로 여러 줄 추가하는 방식 * → 대시보드의 "+ 데이터 추가" 모달에서 여러 줄 한번에 입력 가능 * * 기존 채널(홈페이지/네이버블로그)은 v7과 동일 (문의수 없음, 수동입력). * * [설치 방법] * 1. 확장 프로그램 > Apps Script → 기존 코드 전부 지우고 이 파일 내용 붙여넣기 → 저장 * 2. 배포 > 배포 관리 > 연필 아이콘 > "새 버전"으로 다시 배포 (URL 그대로 유지) * 3. onEdit 트리거는 그대로 유지 (건드릴 필요 없음) * 4. 링크드인 탭은 이제 안 쓰니 지우셔도 되고 그냥 두셔도 상관없어요. */ const SHEET_CONFIG = { home: { name: '홈페이지', keys: ['date', 'visitors', 'pageviews', 'source'], headers: ['날짜', '방문자수', '조회수', '출처(수동/GA4)'] }, naverblog: { name: '네이버블로그', keys: ['date', 'visitors', 'postClicks', 'comments', 'messages'], headers: ['날짜', '방문자수', '게시물조회수', '댓글', '쪽지'] } }; const INPUT_SOURCE_LABEL = '입력경로'; const SNAPSHOT_CONFIG = { homeCategories: { name: '홈페이지_인기카테고리', itemLabel: '카테고리' }, homeContents: { name: '홈페이지_인기콘텐츠', itemLabel: '콘텐츠명' }, blogContents: { name: '네이버블로그_인기콘텐츠', itemLabel: '콘텐츠명' } }; /* ---------- 조회 ---------- */ function doGet(e) { const result = { home: {}, naverblog: {} }; Object.keys(SHEET_CONFIG).forEach(channel => { const cfg = SHEET_CONFIG[channel]; const sheet = getOrCreateChannelSheet_(channel); const lastRow = sheet.getLastRow(); if (lastRow < 2) return; const values = sheet.getRange(2, 1, lastRow - 1, cfg.keys.length + 1).getValues(); values.forEach(row => { const date = normalizeDate_(row[0]); if (!date) return; const obj = {}; cfg.keys.forEach((k, i) => { if (k !== 'date') obj[k] = row[i]; }); obj.inputSource = row[cfg.keys.length] || ''; result[channel][date] = obj; }); }); Object.keys(SNAPSHOT_CONFIG).forEach(key => { result[key] = readSnapshotSheet_(key); }); return jsonOutput_(result); } /* ---------- 입력/수정/삭제 (대시보드에서 호출) ---------- */ function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const body = JSON.parse(e.postData.contents); if (body.action === 'appendSnapshot') return handleAppendSnapshot_(body); if (body.action === 'deleteSnapshotRow') return handleDeleteSnapshotRow_(body); if (body.action === 'delete') return handleDelete_(body); return handleUpsert_(body); } catch (err) { return jsonOutput_({ ok: false, error: err.message }); } finally { lock.releaseLock(); } } function handleUpsert_(body) { const cfg = SHEET_CONFIG[body.channel]; if (!cfg) return jsonOutput_({ ok: false, error: '알 수 없는 채널: ' + body.channel }); const sheet = getOrCreateChannelSheet_(body.channel); const date = normalizeDate_(body.date); const rowIndex = findRow_(sheet, date); const targetRow = rowIndex > 0 ? rowIndex : sheet.getLastRow() + 1; const rowValues = cfg.keys.map(k => { if (k === 'date') return date; return body[k] !== undefined ? body[k] : ''; }); sheet.getRange(targetRow, 1, 1, rowValues.length).setValues([rowValues]); sheet.getRange(targetRow, 1).setNumberFormat('@').setValue(date); sheet.getRange(targetRow, cfg.keys.length + 1).setValue('대시보드'); return jsonOutput_({ ok: true, action: rowIndex > 0 ? 'updated' : 'appended' }); } function handleDelete_(body) { const cfg = SHEET_CONFIG[body.channel]; if (!cfg) return jsonOutput_({ ok: false, error: '알 수 없는 채널: ' + body.channel }); const sheet = getOrCreateChannelSheet_(body.channel); const rowIndex = findRow_(sheet, normalizeDate_(body.date)); if (rowIndex > 0) sheet.deleteRow(rowIndex); return jsonOutput_({ ok: true, action: 'deleted' }); } /* ---------- 스냅샷(인기카테고리/콘텐츠) 입력/삭제 ---------- */ function handleAppendSnapshot_(body) { const cfg = SNAPSHOT_CONFIG[body.channel]; if (!cfg) return jsonOutput_({ ok: false, error: '알 수 없는 스냅샷 채널: ' + body.channel }); const sheet = getOrCreateSnapshotSheet_(body.channel); const date = normalizeDate_(body.date); sheet.appendRow([date, body.item, Number(body.views || 0)]); sheet.getRange(sheet.getLastRow(), 1).setNumberFormat('@').setValue(date); return jsonOutput_({ ok: true, action: 'appended' }); } function handleDeleteSnapshotRow_(body) { const cfg = SNAPSHOT_CONFIG[body.channel]; if (!cfg) return jsonOutput_({ ok: false, error: '알 수 없는 스냅샷 채널: ' + body.channel }); const sheet = getOrCreateSnapshotSheet_(body.channel); const lastRow = sheet.getLastRow(); if (lastRow < 2) return jsonOutput_({ ok: true, action: 'noop' }); const values = sheet.getRange(2, 1, lastRow - 1, 2).getValues(); const targetDate = normalizeDate_(body.date); for (let i = 0; i < values.length; i++) { if (normalizeDate_(values[i][0]) === targetDate && String(values[i][1]) === String(body.item)) { sheet.deleteRow(i + 2); return jsonOutput_({ ok: true, action: 'deleted' }); } } return jsonOutput_({ ok: true, action: 'notfound' }); } function readSnapshotSheet_(key) { const cfg = SNAPSHOT_CONFIG[key]; const sheet = getOrCreateSnapshotSheet_(key); const rows = []; const lastRow = sheet.getLastRow(); if (lastRow < 2) return rows; const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues(); values.forEach(row => { const date = normalizeDate_(row[0]); if (!date || row[1] === '') return; rows.push({ date, item: row[1], views: Number(row[2] || 0) }); }); return rows; } function getOrCreateSnapshotSheet_(key) { const cfg = SNAPSHOT_CONFIG[key]; const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(cfg.name); if (!sheet) { sheet = ss.insertSheet(cfg.name); sheet.appendRow(['날짜', cfg.itemLabel, '조회수']); sheet.setFrozenRows(1); sheet.getRange(2, 1, 999, 1).setNumberFormat('@'); } return sheet; } /* ---------- 시트에서 사람이 직접 수정하면 자동으로 "시트"라고 표시 ---------- */ function onEdit(e) { try { const sheet = e.range.getSheet(); const channel = Object.keys(SHEET_CONFIG).find(ch => SHEET_CONFIG[ch].name === sheet.getName()); if (!channel) return; const cfg = SHEET_CONFIG[channel]; const inputSourceCol = cfg.keys.length + 1; const startRow = e.range.getRow(); const endRow = e.range.getLastRow(); const startCol = e.range.getColumn(); const endCol = e.range.getLastColumn(); if (startCol <= inputSourceCol && endCol >= inputSourceCol && startCol === endCol) return; for (let r = startRow; r <= endRow; r++) { if (r < 2) continue; sheet.getRange(r, inputSourceCol).setValue('시트'); } } catch (err) { // 조용히 무시 } } function getOrCreateChannelSheet_(channel) { const cfg = SHEET_CONFIG[channel]; const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(cfg.name); const fullHeader = cfg.headers.concat([INPUT_SOURCE_LABEL]); if (!sheet) { sheet = ss.insertSheet(cfg.name); sheet.appendRow(fullHeader); sheet.setFrozenRows(1); sheet.getRange(2, 1, 999, 1).setNumberFormat('@'); return sheet; } // 헤더 행이 예전 구조(문의수 열, 중복 입력경로 열 등)로 어긋나 있을 수 있으므로 항상 정확히 맞춰줌 const currentHeader = sheet.getRange(1, 1, 1, fullHeader.length).getValues()[0]; const mismatch = fullHeader.some((h, i) => currentHeader[i] !== h); if (mismatch) { sheet.getRange(1, 1, 1, fullHeader.length).setValues([fullHeader]); } return sheet; } function findRow_(sheet, date) { const lastRow = sheet.getLastRow(); if (lastRow < 2) return -1; const dates = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (let i = 0; i < dates.length; i++) { if (normalizeDate_(dates[i][0]) === date) return i + 2; } return -1; } function normalizeDate_(v) { if (!v) return ''; if (v instanceof Date) return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd'); return String(v).trim(); } function jsonOutput_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
